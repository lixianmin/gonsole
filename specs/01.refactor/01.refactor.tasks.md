```go
func TestSignAndParse(t *testing.T) {
    type args struct {
        secretKey  string
        claims     jwt.MapClaims
        expiration time.Duration
    }

    tests := []struct {
        name      string
        args      args
        wantErr   bool
        sleep     time.Duration
    }{
        {"正常签名解析", args{"key", jwt.MapClaims{"id": 123}, time.Hour}, false, 0},
        {"空密钥", args{"", jwt.MapClaims{"id": 123}, time.Hour}, true, 0},
        {"过期令牌", args{"key", jwt.MapClaims{"id": 123}, time.Millisecond}, true, 10 * time.Millisecond},
    }

    for _, tt := range tests {
        t.Run(tt.name, func(t *testing.T) {
            signed, err := Sign(tt.args.secretKey, tt.args.claims, WithExpiration(tt.args.expiration))
            if (err != nil) != tt.wantErr {
                t.Fatalf("Sign() error = %v, wantErr %v", err, tt.wantErr)
            }
            if tt.wantErr {
                return
            }

            if tt.sleep > 0 {
                time.Sleep(tt.sleep)
            }

            _, err = Parse(tt.args.secretKey, signed)
            if (err != nil) != (tt.sleep > 0) {
                t.Errorf("Parse() error = %v, wantErr %v", err, tt.sleep > 0)
            }
        })
    }
}
```
